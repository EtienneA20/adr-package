# Routes publiques avec `Public`

Dans une application où toutes les routes sont protégées par un guard global, le décorateur
`Public()` marque les exceptions (santé, connexion, catalogue…). Il pose la métadonnée
`IS_PUBLIC_KEY`, que le guard lit avec le `Reflector`.

## Exemple : application fictive « Bibliothèque »

Les extraits ci-dessous viennent de l'[application d'exemple](./example-app), compilée et
testée à chaque intégration.

### 1. Le guard global lit la métadonnée

{/* snippet: apps/example-nest/src/auth/auth.guard.ts#guard */}

```ts title="src/auth/auth.guard.ts"
import { IS_PUBLIC_KEY } from "@etiennea20/adr-nest";
import { type CanActivate, type ExecutionContext, Injectable } from "@nestjs/common";
import { Reflector } from "@nestjs/core";

interface IAuthenticatedRequest {
  user?: unknown;
}

@Injectable()
export class AuthGuard implements CanActivate {
  constructor(private readonly reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const isPublic = this.reflector.getAllAndOverride<boolean | undefined>(IS_PUBLIC_KEY, [
      context.getHandler(),
      context.getClass(),
    ]);
    if (isPublic) return true;
    return Boolean(context.switchToHttp().getRequest<IAuthenticatedRequest>().user);
  }
}
```

### 2. Le guard est enregistré globalement

{/* snippet: apps/example-nest/src/app.module.ts#app-module */}

```ts title="src/app.module.ts"
import { Module } from "@nestjs/common";
import { APP_GUARD, APP_PIPE } from "@nestjs/core";

import { AuthGuard } from "./auth";
import { BooksModule } from "./books";
import { HealthController } from "./health";
import { nativeValidationPipe } from "./validation";

@Module({
  imports: [BooksModule],
  controllers: [HealthController],
  providers: [
    { provide: APP_GUARD, useClass: AuthGuard },
    { provide: APP_PIPE, useValue: nativeValidationPipe },
  ],
})
export class AppModule {}
```

### 3. Les routes publiques sont marquées

{/* snippet: apps/example-nest/src/books/books.controller.ts#public-routes */}

```ts title="src/books/books.controller.ts"
import type { IPage } from "@etiennea20/adr-misc/contract";
import { Public, ValidBody } from "@etiennea20/adr-nest";
import { type IListCriteria, ListQuery } from "@etiennea20/adr-nest/filtering";
import { Body, Controller, Get, Param, Patch, Post } from "@nestjs/common";

import type { IBook } from "./book";
import { bookListQuery } from "./book-list-query";
import {
  type CreateBookInput,
  type UpdateBookInput,
  createBookSchema,
  updateBookSchema,
} from "./book.schemas";
import { BooksService } from "./books.service";

@Controller("livres")
export class BooksController {
  constructor(private readonly books: BooksService) {}

  @Public() // le catalogue est consultable sans compte
  @Get() // GET /livres?filter=…&sort=…&page=2 (un paramètre par clé, en JSON)
  list(@ListQuery(bookListQuery) criteria: IListCriteria): IPage<IBook> {
    return this.books.findPage(criteria);
  }

  @Public()
  @Post("search") // même contrat dans le corps, pour les filtres trop longs pour une URL
  search(@ListQuery(bookListQuery, { source: "body" }) criteria: IListCriteria): IPage<IBook> {
    return this.books.findPage(criteria);
  }

  @Get(":id") // le détail (avec l'emprunteur) est réservé aux utilisateurs connectés
  get(@Param("id") id: string): IBook {
    return this.books.get(id);
  }

  @Post() // NestJS 11 et 12 : pipe du package, erreurs au format d'API
  create(@ValidBody(createBookSchema) input: CreateBookInput): IBook {
    return this.books.create(input);
  }

  @Patch(":id") // NestJS 12 : validation native (pipe global de AppModule)
  update(
    @Param("id") id: string,
    @Body({ schema: updateBookSchema }) input: UpdateBookInput,
  ): IBook {
    return this.books.update(id, input);
  }
}
```

Décorer la classe (`@Public()` au-dessus de `@Controller`) rend publiques toutes ses routes :

{/* snippet: apps/example-nest/src/health/health.controller.ts#public-controller */}

```ts title="src/health/health.controller.ts"
@Public() // toutes les routes du contrôleur sont publiques
@Controller("health")
export class HealthController {
  @Get()
  check(): { status: "ok" } {
    return { status: "ok" };
  }
}
```
